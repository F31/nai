cmake_minimum_required(VERSION 3.20)
project(nai_contract_roundtrip LANGUAGES CXX)
find_package(Protobuf 3.21.12 EXACT REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE CONTRACT_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/contracts/generated/*.pb.cc")
add_executable(contract-roundtrip main.cpp ${CONTRACT_SOURCES})
target_include_directories(contract-roundtrip PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/contracts/generated")
target_link_libraries(contract-roundtrip PRIVATE protobuf::libprotobuf)
